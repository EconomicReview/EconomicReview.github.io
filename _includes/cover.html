{%- comment -%} Generated typographic cover. Usage: {% include cover.html pub=p heading="h3" %} → cover with title {% include cover.html pub=p decorative=true %} → motif-only thumb (hidden from AT; pair with a real text link) A real image in `cover:` front matter overrides the generated cover. The motif is chosen by hashing the slug, so a cover never changes between builds. The title is real HTML text (not SVG), so it wraps, scales, and is read by assistive tech; the motif SVG is decorative and aria-hidden. {%- endcomment -%} {%- assign slug = include.pub.url | split: "/" | last -%} {%- assign h = 0 -%} {%- assign chars = slug | split: "" -%} {%- for c in chars -%} {%- case c -%} {%- when "b","h","n","t","z","5" -%}{%- assign h = h | plus: 1 -%} {%- when "c","i","o","u","0","6" -%}{%- assign h = h | plus: 2 -%} {%- when "d","j","p","v","1","7" -%}{%- assign h = h | plus: 3 -%} {%- when "e","k","q","w","2","8" -%}{%- assign h = h | plus: 4 -%} {%- when "f","l","r","x","3","9" -%}{%- assign h = h | plus: 5 -%} {%- endcase -%} {%- endfor -%} {%- assign motif = h | modulo: 6 -%} {%- assign track = include.pub.track | default: "research" -%} {%- capture kicker -%} {%- if track == "research" -%} {%- if include.pub.volume -%}Vol. {{ include.pub.volume }}{% if include.pub.issue %}, {{ include.pub.issue }}{% endif %}{%- else -%}Research{%- endif -%} {%- else -%} Perspectives, {{ include.pub.date | date: "%Y" }} {%- endif -%} {%- endcapture -%} {%- if include.pub.cover and include.pub.cover != "" -%} {%- else -%} {%- case motif -%} {%- when 0 -%} {%- when 1 -%} {%- when 2 -%} {%- when 3 -%} {%- when 4 -%} {%- when 5 -%} {%- endcase -%} {{ kicker | strip }} {%- unless include.decorative -%} {%- assign tag = include.heading | default: "span" -%} <{{ tag }} class="cover-title">{{ include.pub.title }} {%- endunless -%} {%- endif -%}
